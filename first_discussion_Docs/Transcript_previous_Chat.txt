#!/usr/bin/env python3
import json
import re
import base64
from datetime import datetime

def extract_blobs_from_jsonl(filepath):
    """Extract all blobs from archived network traffic"""
    
    blobs = {
        'static_html_blobs': [],
        'api_request_blobs': [],
        'sdk_data_blobs': [],
        'verification_blobs': []
    }
    
    with open(filepath, 'r') as f:
        for line_num, line in enumerate(f, 1):
            try:
                entry = json.loads(line.strip())
                timestamp = entry.get('timestamp', 'UNKNOWN')
                url = entry.get('url', '')
                
                # 1. Extract blob from HTML response (generating.html)
                if 'generating.html' in url or 'karakoselabs.com' in url:
                    response_body = entry.get('response_body', '')
                    
                    # Pattern: data: { blob: "BASE64_STRING" }
                    html_blob_match = re.search(
                        r'data:\s*{\s*blob:\s*["\']([A-Za-z0-9+/=]+)["\']',
                        response_body
                    )
                    if html_blob_match:
                        blob_value = html_blob_match.group(1)
                        blobs['static_html_blobs'].append({
                            'timestamp': timestamp,
                            'url': url,
                            'blob': blob_value,
                            'line': line_num,
                            'context': 'HTML_EMBEDDED'
                        })
                        print(f"[+] Found HTML blob at line {line_num}")
                        print(f"    Length: {len(blob_value)} chars")
                        print(f"    Preview: {blob_value[:60]}...")
                
                # 2. Extract blob from POST request body
                request_body = entry.get('request_body', '')
                
                # JSON request body
                if isinstance(request_body, dict):
                    if 'blob' in request_body:
                        blobs['api_request_blobs'].append({
                            'timestamp': timestamp,
                            'url': url,
                            'blob': request_body['blob'],
                            'line': line_num,
                            'full_payload': request_body
                        })
                        print(f"[+] Found API request blob at line {line_num}")
                    
                    if 'data' in request_body and isinstance(request_body['data'], dict):
                        if 'blob' in request_body['data']:
                            blobs['api_request_blobs'].append({
                                'timestamp': timestamp,
                                'url': url,
                                'blob': request_body['data']['blob'],
                                'line': line_num,
                                'full_payload': request_body
                            })
                            print(f"[+] Found nested data blob at line {line_num}")
                
                # String request body (might be JSON string)
                elif isinstance(request_body, str):
                    try:
                        parsed = json.loads(request_body)
                        if 'blob' in parsed:
                            blobs['api_request_blobs'].append({
                                'timestamp': timestamp,
                                'url': url,
                                'blob': parsed['blob'],
                                'line': line_num,
                                'full_payload': parsed
                            })
                        if 'data' in parsed and 'blob' in parsed.get('data', {}):
                            blobs['api_request_blobs'].append({
                                'timestamp': timestamp,
                                'url': url,
                                'blob': parsed['data']['blob'],
                                'line': line_num,
                                'full_payload': parsed
                            })
                    except:
                        # Regex fallback for non-JSON strings
                        blob_match = re.search(r'"blob"\s*:\s*"([^"]+)"', request_body)
                        if blob_match:
                            blobs['api_request_blobs'].append({

#!/usr/bin/env python3
import json
from http.cookies import SimpleCookie

def extract_cookies_from_jsonl(filepath):
    """Extract all cookies from network traffic"""
    
    cookie_timeline = []
    unique_cookies = {}
    
    with open(filepath, 'r') as f:
        for line_num, line in enumerate(f, 1):
            try:
                entry = json.loads(line.strip())
                headers = entry.get('headers', {})
                url = entry.get('url', '')
                timestamp = entry.get('timestamp', 0)
                
                # Extract Set-Cookie from responses
                if 'karakoselabs.com' in url:
                    set_cookie = headers.get('set-cookie') or headers.get('Set-Cookie')
                    if set_cookie:
                        if isinstance(set_cookie, list):
                            set_cookie = '; '.join(set_cookie)
                        
                        # Parse cookies
                        cookie = SimpleCookie()
                        cookie.load(set_cookie)
                        
                        for key, morsel in cookie.items():
                            cookie_timeline.append({
                                'timestamp': timestamp,
                                'name': key,
                                'value': morsel.value,
                                'domain': morsel.get('domain', ''),
                                'path': morsel.get('path', '/'),
                                'secure': morsel.get('secure', False),
                                'httponly': morsel.get('httponly', False),
                                'line': line_num,
                                'url': url
                            })
                            
                            # Track latest value for each cookie
                            unique_cookies[key] = {
                                'value': morsel.value,
                                'timestamp': timestamp,
                                'attributes': {
                                    'domain': morsel.get('domain', ''),
                                    'path': morsel.get('path', '/'),
                                    'secure': morsel.get('secure', False),
                                    'httponly': morsel.get('httponly', False)
                                }
                            }
                            
                            print(f"[+] Cookie '{key}' found at line {line_num}")
                            if '__krkse' in key:
                                print(f"    [!] CRITICAL: Karakose tracking cookie detected")
                                print(f"    Value: {morsel.value[:50]}...")
                
                # Extract Cookie from requests
                cookie_header = headers.get('cookie') or headers.get('Cookie')
                if cookie_header and 'karakoselabs.com' in url:
                    print(f"[+] Request cookies at line {line_num}: {cookie_header[:100]}...")
            
            except Exception as e:
                continue
    
    return cookie_timeline, unique_cookies

        print(f"\n✓ Valid base64 encoding")
        print(f"  Decoded size: {len(decoded)} bytes")

    except Exception as e:
